Step 1: Import libraries and load the dataset

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
# Load the Titanic dataset bundled with Seaborn
df = sns.load_dataset('titanic')
print('Shape of dataset:', df.shape)
print('\nFirst 5 records:')
print(df.head())
print('\nDataset Info:')
print(df.info())

Step 2: Compute summary statistics
# Summary statistics for numerical columns
print('Summary statistics:')
print(df.describe())
# Individual measures for the Age column
print('\nMean Age:', df['age'].mean())
print('Median Age:', df['age'].median())
print('Mode Age:', df['age'].mode()[0])
print('Standard Deviation of Age:', df['age'].std())
print('Variance of Age:', df['age'].var())
print('Skewness of Age:', df['age'].skew())
print('Kurtosis of Age:', df['age'].kurt())
# Frequency distribution of a categorical column
print('\nPassenger class value counts:')
print(df['class'].value_counts(dropna=False))
Step 3: Visualize missing values
plt.figure(figsize=(8, 4))
sns.heatmap(df.isnull(), cbar=False, cmap='viridis')
plt.title('Missing Value Map')
plt.tight_layout()
plt.show()


Step 4: Univariate analysis — histograms and count plots
# Histogram of a numerical column
plt.figure(figsize=(6, 4))
sns.histplot(df['age'].dropna(), kde=True, bins=20, color='steelblue')
plt.title('Distribution of Age')
plt.xlabel('Age')
plt.tight_layout()
plt.show()
# Count plot of a categorical column
plt.figure(figsize=(6, 4))
sns.countplot(x='class', data=df, palette='Set2')
plt.title('Passenger Count by Class')
plt.tight_layout()
plt.show()


Step 5: Univariate outlier detection — box plots
plt.figure(figsize=(6, 4))
sns.boxplot(y=df['fare'], color='lightcoral')
plt.title('Box Plot of Fare (Outlier Detection)')
plt.tight_layout()
plt.show()


Step 6: Bivariate analysis — scatter plot and grouped bar chart
# Scatter plot between two numerical variables
plt.figure(figsize=(6, 4))
sns.scatterplot(x='age', y='fare', hue='survived', data=df, s=60)
plt.title('Age vs Fare by Survival')
plt.tight_layout()
plt.show()
# Survival rate by passenger class
survival_rate = df.groupby('class')['survived'].mean()
plt.figure(figsize=(6, 4))
survival_rate.plot(kind='bar', color='mediumseagreen')
plt.title('Survival Rate by Passenger Class')
plt.ylabel('Survival Rate')
plt.tight_layout()
plt.show()


Step 7: Multivariate analysis — correlation heatmap and pair plot
# Correlation heatmap among numerical columns
numeric_df = df.select_dtypes(include=[np.number])
plt.figure(figsize=(6, 5))
sns.heatmap(numeric_df.corr(), annot=True, cmap='coolwarm', fmt='.2f')
plt.title('Correlation Heatmap')
plt.tight_layout()
plt.show()
# Pair plot across selected numerical columns
cols = ['survived', 'age', 'fare', 'sibsp', 'parch']
sns.pairplot(df[cols].dropna(), hue='survived')
plt.suptitle('Pair Plot of Numerical Features', y=1.02)
plt.show()
